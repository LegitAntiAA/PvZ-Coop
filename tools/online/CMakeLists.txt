cmake_minimum_required(VERSION 3.23)
project(PvZOnlineDependencies LANGUAGES C CXX)

if(NOT DEFINED PVZ_VENDOR_DIR)
  message(FATAL_ERROR "Set PVZ_VENDOR_DIR to the pinned dependency checkouts")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
add_subdirectory("${PVZ_VENDOR_DIR}/mbedtls" mbedtls EXCLUDE_FROM_ALL)
add_library(MbedTLS::MbedTLS ALIAS mbedtls)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PVZ_VENDOR_DIR}/libdatachannel" libdatachannel)

if(PVZ_BUILD_ONLINE_TESTS)
  add_executable(NetOnlineTests "${CMAKE_CURRENT_LIST_DIR}/../../tests/Online/NetOnlineTests.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../../Lawn/Net/NetOnline.cpp")
  target_compile_features(NetOnlineTests PRIVATE cxx_std_20)
  target_compile_definitions(NetOnlineTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_custom_command(TARGET NetOnlineTests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:datachannel>" "$<TARGET_FILE_DIR:NetOnlineTests>")
  add_dependencies(NetOnlineTests datachannel)
endif()
